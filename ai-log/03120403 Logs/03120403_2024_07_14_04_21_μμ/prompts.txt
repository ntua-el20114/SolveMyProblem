    res.status(400).json({ error: error.message });
  }
});

app.get('/problems', async (req, res) => {
  try {
    const problems = await Problem.findAll();
    res.status(200).json(problems);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
